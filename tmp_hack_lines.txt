39: /* ─── Image Carousel ────────────────────────────────────── */
40: function ImageCarousel() {
41:     const [current, setCurrent] = useState(0);
42:     const [direction, setDirection] = useState(0);
43:     const touchStart = useRef(null);
44:     const len = galleryImages.length;
45: 
46:     const go = useCallback((dir) => {
47:         setDirection(dir);
48:         setCurrent(prev => (prev + dir + len) % len);
49:     }, [len]);
50: 
51:     useEffect(() => {
52:         const handler = (e) => {
53:             if (e.key === 'ArrowLeft') go(-1);
54:             if (e.key === 'ArrowRight') go(1);
55:         };
56:         window.addEventListener('keydown', handler);
57:         return () => window.removeEventListener('keydown', handler);
58:     }, [go]);
59: 
60:     const variants = {
61:         enter: (dir) => ({ x: dir > 0 ? 300 : -300, opacity: 0 }),
62:         center: { x: 0, opacity: 1 },
63:         exit: (dir) => ({ x: dir > 0 ? -300 : 300, opacity: 0 }),
64:     };
65: 
66:     const img = galleryImages[current];
67: 
68:     return (
69:         <div className="relative w-full">
70:             {/* Main image container */}
71:             <div
72:                 className="relative bg-neutral-100 overflow-hidden aspect-video border border-black/10 rounded-[2px]"
73:                 onTouchStart={(e) => { touchStart.current = e.touches[0].clientX; }}
74:                 onTouchEnd={(e) => {
75:                     if (!touchStart.current) return;
76:                     const diff = e.changedTouches[0].clientX - touchStart.current;
77:                     if (Math.abs(diff) > 50) go(diff < 0 ? 1 : -1);
78:                     touchStart.current = null;
79:                 }}
80:             >
81:                 <GsapPresence initial={false} custom={direction} mode="wait">
82:                     <Gsap.div
83:                         key={current}
84:                         custom={direction}
85:                         variants={variants}
86:                         initial="enter"
87:                         animate="center"
88:                         exit="exit"
89:                         transition={{ duration: 0.35, ease: [0.33, 1, 0.68, 1] }}
90:                         className="absolute inset-0"
91:                     >
92:                         <img
93:                             src={img.src}
94:                             alt={img.alt}
95:                             loading="lazy"
96:                             decoding="async"
97:                             className="w-full h-full object-contain"
98:                         />
99:                     </Gsap.div>
100:                 </GsapPresence>
101: 
102:                 {/* Nav arrows - Minimal styling */}
103:                 <button
104:                     onClick={(e) => { e.stopPropagation(); go(-1); }}
105:                     className="absolute left-4 top-1/2 -translate-y-1/2 z-10 w-10 h-10 bg-white/50 backdrop-blur-md rounded-full shadow-sm flex items-center justify-center hover:bg-white text-black/60 hover:text-black transition-all"
106:                     aria-label="Previous image"
107:                 >
108:                     <ChevronLeft size={20} strokeWidth={2} />
109:                 </button>
110:                 <button
111:                     onClick={(e) => { e.stopPropagation(); go(1); }}
112:                     className="absolute right-4 top-1/2 -translate-y-1/2 z-10 w-10 h-10 bg-white/50 backdrop-blur-md rounded-full shadow-sm flex items-center justify-center hover:bg-white text-black/60 hover:text-black transition-all"
113:                     aria-label="Next image"
114:                 >
115:                     <ChevronRight size={20} strokeWidth={2} />
116:                 </button>
117: 
118:                 {/* Counter */}
119:                 <div className="absolute bottom-4 right-4 z-10 bg-black/40 backdrop-blur-md text-white px-3 py-1.5 rounded-full font-mono text-[10px] tracking-[0.12em] md:tracking-[0.16em] backdrop-saturate-150">
120:                     {String(current + 1).padStart(2, '0')} / {String(len).padStart(2, '0')}
121:                 </div>
122:             </div>
123: 
124:             {/* Thumbnail strip */}
125:             <div className="flex gap-2 mt-4 overflow-x-auto pb-2 scrollbar-hide">
126:                 {galleryImages.map((img, i) => (
127:                     <button
128:                         key={i}
129:                         onClick={() => { setDirection(i > current ? 1 : -1); setCurrent(i); }}
130:                         className={`
131:                             shrink-0 w-20 h-14 md:w-24 md:h-16 rounded-[2px] overflow-hidden transition-all duration-300 relative
132:                             ${i === current ? 'opacity-100 ring-2 ring-black ring-offset-2' : 'opacity-40 hover:opacity-100'}
133:                         `}
134:                     >
135:                         <img src={img.src} alt="" className="w-full h-full object-cover" loading="lazy" />
136:                     </button>
137:                 ))}
138:             </div>
139:         </div>
140:     );
141: }
